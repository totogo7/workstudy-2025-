```java
package com.library.common;

import lombok.Data;

@Data
public class Result<T> {
    private Integer code;
    private String msg;
    private T data;

    public static <T> Result<T> success(T data) {
        Result<T> r = new Result<>();
        r.setCode(1);
        r.setMsg("success");
        r.setData(data);
        return r;
    }
    public static <T> Result<T> error(String msg) {
        Result<T> r = new Result<>();
        r.setCode(0);
        r.setMsg(msg);
        return r;
    }
}
```

## 3) Flyway 建表脚本（阶段 2 同时）

放 `library-backend/src/main/resources/db/migration/V1__init.sql`（Flyway 会自动执行）

```sql
-- V1__init.sql
CREATE TABLE IF NOT EXISTS users (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(100) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL,
  role VARCHAR(50) DEFAULT 'user',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS books (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  isbn VARCHAR(20),
  title VARCHAR(255) NOT NULL,
  summary TEXT,
  publisher VARCHAR(128),
  publish_date DATE,
  total_copies INT DEFAULT 0,
  available_copies INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS book_copies (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  book_id BIGINT NOT NULL,
  barcode VARCHAR(64),
  copy_status TINYINT DEFAULT 1,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_copy_book FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS borrow_records (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  copy_id BIGINT NOT NULL,
  book_id BIGINT NOT NULL,
  borrow_time DATETIME NOT NULL,
  due_time DATETIME NOT NULL,
  return_time DATETIME NULL,
  status TINYINT DEFAULT 1,
  CONSTRAINT fk_br_copy FOREIGN KEY (copy_id) REFERENCES book_copies(id) ON DELETE RESTRICT
);

CREATE TABLE IF NOT EXISTS reservations (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  book_id BIGINT NOT NULL,
  reserve_time DATETIME DEFAULT CURRENT_TIMESTAMP,
  status TINYINT DEFAULT 1
);
```

# 四、常见问题 & 排查建议

- **MySQL 连接失败**：确认 docker-compose 中端口是否被占用；检查 `docker logs lib_mysql`；如果使用 WSL2，`localhost` 通常可连，但在某些环境需使用 `127.0.0.1` 或宿主机 IP。
- **Flyway 未执行脚本**：确认 `application.yml` 中 `spring.flyway.locations` 配置正确，脚本放在 `classpath:db/migration` 且文件名以 `V1__` 开头。检查启动日志看 Flyway 日志。
- **端口冲突**：若 8080/3306 被占用，修改 `application.yml` 或 `docker-compose.yml` 映射端口。
- **IDEA 编译报 Lombok 错误**：在 IDEA 安装 Lombok 插件并开启 annotation processing。
