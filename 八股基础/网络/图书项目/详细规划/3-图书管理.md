```java
import com.library.entity.Book;

public interface BookService {

    Book update(Long id, Book book);
    boolean delete(Long id);
```

实现：`library-backend/src/main/java/com/library/service/impl/BookServiceImpl.java`

```java
package com.library.service.impl;

import com.github.pagehelper.PageHelper;
import com.github.pagehelper.PageInfo;
import com.library.entity.Book;
import com.library.mapper.BookMapper;
import com.library.service.BookService;
import org.springframework.stereotype.Service;
import javax.annotation.Resource;
import java.util.List;

@Service
public class BookServiceImpl implements BookService {

    @Resource
    private BookMapper bookMapper;

    @Override
    public PageInfo<Book> list(String title, String author, String isbn, int page, int size) {
        PageHelper.startPage(page, size);
        List<Book> list = bookMapper.search(title, author, isbn);
        return new PageInfo<>(list);
    }

    @Override
    public Book getById(Long id) {
        return bookMapper.findById(id);
    }

    @Override
    public Book create(Book book) {
        if (book.getTotalCopies() == null) book.setTotalCopies(0);
        if (book.getAvailableCopies() == null) book.setAvailableCopies(book.getTotalCopies());
        bookMapper.insert(book);
        return book;
    }

    @Override
    public Book update(Long id, Book book) {
        book.setId(id);
        bookMapper.update(book);
        return bookMapper.findById(id);
    }

    @Override
    public boolean delete(Long id) {
        return bookMapper.deleteById(id) == 1;
    }
}
```

## 文件 6 — Controller：BookController.java

路径：`library-backend/src/main/java/com/library/controller/BookController.java`

```java
package com.library.controller;

import com.library.common.Result;
import com.library.dto.BookDto;
import com.library.entity.Book;
import com.library.service.BookService;
import com.github.pagehelper.PageInfo;
import org.springframework.web.bind.annotation.*;

import javax.annotation.Resource;
import javax.validation.Valid;

@RestController
@RequestMapping("/api/books")
public class BookController {

    @Resource
    private BookService bookService;

    @GetMapping
    public Result<PageInfo<Book>> list(@RequestParam(required = false) String title,
                                       @RequestParam(required = false) String author,
                                       @RequestParam(required = false) String isbn,
                                       @RequestParam(defaultValue = "1") int page,
                                       @RequestParam(defaultValue = "10") int size) {
        PageInfo<Book> info = bookService.list(title, author, isbn, page, size);
        return Result.success(info);
    }

    @GetMapping("/{id}")
    public Result<Book> get(@PathVariable Long id) {
        Book b = bookService.getById(id);
        if (b == null) return Result.error("not found");
        return Result.success(b);
    }

    @PostMapping
    public Result<Book> create(@Valid @RequestBody BookDto dto) {
        Book b = dtoToEntity(dto);
        Book created = bookService.create(b);
        return Result.success(created);
    }

    @PutMapping("/{id}")
    public Result<Book> update(@PathVariable Long id, @Valid @RequestBody BookDto dto) {
        Book b = dtoToEntity(dto);
        Book updated = bookService.update(id, b);
        return Result.success(updated);
    }

    @DeleteMapping("/{id}")
    public Result<String> delete(@PathVariable Long id) {
        boolean ok = bookService.delete(id);
        if (!ok) return Result.error("delete failed");
        return Result.success("deleted");
    }

    private Book dtoToEntity(BookDto dto) {
        Book b = new Book();
        b.setTitle(dto.getTitle());
        b.setAuthor(dto.getAuthor());
        b.setIsbn(dto.getIsbn());
        b.setSummary(dto.getSummary());
        b.setPublisher(dto.getPublisher());
        b.setPublishDate(dto.getPublishDate());
        b.setTotalCopies(dto.getTotalCopies());
        b.setAvailableCopies(dto.getTotalCopies()); // default same
        return b;
    }
}
```

```vue
<template>
  <div>

      <el-table-column prop="title" label="书名"/>

          <el-button size="small" @click="edit(row)">编辑</el-button>
          <el-button size="small" type="danger" @click="del(row)">删除</el-button>
          <el-button size="small" type="primary" @click="borrow(row)">借书</el-button>
        </template>
      </el-table-column>
    </el-table>

    <div class="pager" style="margin-top: 16px;">
      <el-pagination
        background
        layout="prev, pager, next, jumper, ->, total"
        :page-size="pageSize"
        :current-page="page"
        :total="total"
        @current-change="onPageChange"/>
    </div>

    <el-dialog :visible.sync="showEdit" title="新增/编辑图书" width="600px">
      <book-edit :book="editing" @saved="onSaved" @cancel="showEdit=false" />
    </el-dialog>
  </div>
</template>

<script>
import axios from 'axios'
import BookEdit from '@/components/BookEdit.vue'

export default {
  components: { BookEdit },
  data() {
    return {
      filters: { title: '', author: '' },
      books: [],
      page: 1,
      pageSize: 10,
      total: 0,
      loading: false,
      showEdit: false,
      editing: null
    }
  },
  methods: {
    async load() {
      this.loading = true;
      try {
        const res = await axios.get('/api/books', { params: { ...this.filters, page: this.page, size: this.pageSize }});
        if (res.data.code === 1) {
          const info = res.data.data;
          this.books = info.list;
          this.total = info.total;
        } else {
          this.$message.error(res.data.msg);
        }
      } finally { this.loading = false; }
    },
    onSearch() { this.page = 1; this.load(); },
    onPageChange(p) { this.page = p; this.load(); },
    openCreate() { this.editing = null; this.showEdit = true; },
    edit(row) { this.editing = row; this.showEdit = true; },
    async del(row) {
      await this.$confirm('确认删除？','提示');
      const res = await axios.delete(`/api/books/${row.id}`);
      if (res.data.code === 1) { this.$message.success('删除成功'); this.load(); } else this.$message.error(res.data.msg);
    },
    borrow(row) {
      // 跳转借书流程（把后端借书接口实现后接入）
      this.$router.push({ name: 'Borrow', query: { bookId: row.id }});
    },
    onSaved() { this.showEdit = false; this.load(); }
  },
  mounted() { this.load(); }
}
</script>
```

## 文件 8 — BookEdit.vue (组件)

路径：`library-frontend/src/components/BookEdit.vue`

```vue
<template>
  <el-form :model="form" :rules="rules" ref="formRef" label-width="100px">
    <el-form-item label="书名" prop="title"><el-input v-model="form.title"></el-input></el-form-item>
    <el-form-item label="作者" prop="author"><el-input v-model="form.author"></el-input></el-form-item>
    <el-form-item label="ISBN" prop="isbn"><el-input v-model="form.isbn"></el-input></el-form-item>
    <el-form-item label="出版社" prop="publisher"><el-input v-model="form.publisher"></el-input></el-form-item>
    <el-form-item label="出版日期" prop="publishDate"><el-date-picker v-model="form.publishDate" type="date" placeholder="选择日期" /></el-form-item>
    <el-form-item label="总副本数" prop="totalCopies"><el-input-number v-model="form.totalCopies" :min="0" /></el-form-item>

    <el-form-item>
      <el-button type="primary" @click="save">保存</el-button>
      <el-button @click="$emit('cancel')">取消</el-button>
    </el-form-item>
  </el-form>
</template>

<script>
import axios from 'axios'
import { ref } from 'vue'

export default {
  props: {
    book: { type: Object, default: null }
  },
  setup(props, { emit }) {
    const formRef = ref(null)
    const form = ref({
      title: '',
      author: '',
      isbn: '',
      publisher: '',
      publishDate: null,
      totalCopies: 0
    })
    const rules = {
      title: [{ required: true, message: '请填写书名', trigger: 'blur' }]
    }
    if (props.book) {
      form.value = {
        title: props.book.title,
        author: props.book.author,
        isbn: props.book.isbn,
        publisher: props.book.publisher,
        publishDate: props.book.publishDate,
        totalCopies: props.book.totalCopies
      }
    }
    const save = async () => {
      formRef.value.validate(async (valid) => {
        if (!valid) return;
        try {
          if (props.book && props.book.id) {
            const res = await axios.put(`/api/books/${props.book.id}`, form.value);
            if (res.data.code === 1) { emit('saved'); this.$message.success('更新成功'); }
            else this.$message.error(res.data.msg);
          } else {
            const res = await axios.post(`/api/books`, form.value);
            if (res.data.code === 1) { emit('saved'); this.$message.success('新增成功'); }
            else this.$message.error(res.data.msg);
          }
        } catch (e) { console.error(e); }
      });
    };
    return { formRef, form, rules, save };
  }
}
</script>
```
