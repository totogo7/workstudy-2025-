CREATE TABLE IF NOT EXISTS book_copies (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  book_id BIGINT NOT NULL,
  barcode VARCHAR(64),
  copy_status TINYINT DEFAULT 1 COMMENT '1=available,2=borrowed,3=reserved',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_copy_book FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS borrow_records (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  user_id BIGINT NOT NULL,
  copy_id BIGINT NOT NULL,
  book_id BIGINT NOT NULL,
  borrow_time DATETIME NOT NULL,
  due_time DATETIME NOT NULL,
  return_time DATETIME NULL,
  status TINYINT DEFAULT 1 COMMENT '1=borrowed,2=returned,3=overdue',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_br_copy FOREIGN KEY (copy_id) REFERENCES book_copies(id) ON DELETE RESTRICT

## 4) Service：事务性借书/还书实现

### `BorrowService.java`

路径：`library-backend/src/main/java/com/library/service/BorrowService.java`

```java
package com.library.service;

public interface BorrowService {
    /**
     * 尝试借书（事务内）
     * @param userId 借书用户
     * @param bookId 要借的书号（book 表 id）
     * @return BorrowRecord id if success, null if no copy available
     */
    Long borrowBook(Long userId, Long bookId);

    /**
     * 还书（根据 borrowRecordId）
     * @param borrowRecordId
     * @return true if success
     */
    boolean returnBook(Long borrowRecordId);
}
```

### `BorrowServiceImpl.java`

路径：`library-backend/src/main/java/com/library/service/impl/BorrowServiceImpl.java`

```java
package com.library.service.impl;

import com.library.entity.BookCopy;
import com.library.entity.BorrowRecord;
import com.library.mapper.BookCopyMapper;
import com.library.mapper.BorrowRecordMapper;
import com.library.mapper.BookMapper;
import com.library.service.BorrowService;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import javax.annotation.Resource;
import java.time.LocalDateTime;

@Service
public class BorrowServiceImpl implements BorrowService {

    @Resource
    private BookCopyMapper bookCopyMapper;
    @Resource
    private BorrowRecordMapper borrowRecordMapper;
    @Resource
    private BookMapper bookMapper;

    // 最大借书数（简单策略）
    private static final int MAX_BORROW = 5;
    private static final int BORROW_DAYS = 30;

    @Override
    @Transactional
    public Long borrowBook(Long userId, Long bookId) {
        // 1. 可选：检查用户当前未归还数量（这里示例不做复杂查询，真实环境请做）
        // 2. select available copy for update
        BookCopy copy = bookCopyMapper.selectAvailableCopyForUpdate(bookId);
        if (copy == null) {
            return null; // no available copy
        }
        // 3. update copy status -> borrowed
        int u = bookCopyMapper.updateCopyStatus(copy.getId(), 2);
        if (u != 1) {
            // someone else updated, treat as failed
            return null;
        }
        // 4. insert borrow record
        BorrowRecord record = new BorrowRecord();
        record.setUserId(userId);
        record.setCopyId(copy.getId());
        record.setBookId(bookId);
        LocalDateTime now = LocalDateTime.now();
        record.setBorrowTime(now);
        record.setDueTime(now.plusDays(BORROW_DAYS));
        record.setStatus(1);
        borrowRecordMapper.insert(record);
        // 5. update book.available_copies = -1
        bookMapper.updateAvailableCopies(bookId, -1);
        return record.getId();
    }

    @Override
    @Transactional
    public boolean returnBook(Long borrowRecordId) {
        // load borrow record (not strictly necessary if we have copyId)
        // For simplicity, we assume borrowRecord exists and has copyId
        // Implementation: update borrow_records return_time, update copy_status -> available, update book.available_copies +1
        // We need a mapper method to find borrow record by id (if not exist, add)
        // For demo let's query minimal via SQL: we'll add a simple mapper call
        BorrowRecord br = borrowRecordMapper.selectById(borrowRecordId);
        if (br == null) return false;
        if (br.getReturnTime() != null) return false; // already returned

        borrowRecordMapper.updateReturnTime(borrowRecordId);
        bookCopyMapper.updateCopyStatus(br.getCopyId(), 1); // available
        bookMapper.updateAvailableCopies(br.getBookId(), 1);
        return true;
    }
}
```

> 备注：上面 `borrowRecordMapper.selectById` 以及 `BorrowRecordMapper` 的 select 方法如果不存在请补上；你也可以在 controller 层直接传 `copyId` 来做 return，逻辑同理。

请在 `BorrowRecordMapper.xml` 添加：

## 5) Controller：借书 / 还书 API

### `BorrowController.java`

路径：`library-backend/src/main/java/com/library/controller/BorrowController.java`

```java
package com.library.controller;

import com.library.common.Result;
import com.library.service.BorrowService;
import org.springframework.web.bind.annotation.*;

import javax.annotation.Resource;

@RestController
@RequestMapping("/api/borrow")
public class BorrowController {

    @Resource
    private BorrowService borrowService;

    // POST /api/borrow?userId=1&bookId=10
    @PostMapping
    public Result<?> borrow(@RequestParam Long userId, @RequestParam Long bookId) {
        Long recordId = borrowService.borrowBook(userId, bookId);
        if (recordId == null) return Result.error("no available copy or failed");
        return Result.success(recordId);
    }

    // POST /api/borrow/return?recordId=123
    @PostMapping("/return")
    public Result<?> ret(@RequestParam Long recordId) {
        boolean ok = borrowService.returnBook(recordId);
        if (!ok) return Result.error("return failed");
        return Result.success("returned");
    }
}
```

> 实务中应从鉴权上下文获取 `userId`，不要在请求里直接传 userId（这里为方便演示与测试我允许传参）。

```java
package com.library;

import com.library.entity.Book;
import com.library.entity.BookCopy;
import com.library.mapper.BookCopyMapper;
import com.library.mapper.BookMapper;
import com.library.service.BorrowService;
import org.junit.jupiter.api.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.testcontainers.containers.MySQLContainer;
import org.testcontainers.junit.jupiter.Container;
import org.testcontainers.junit.jupiter.Testcontainers;

import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

@SpringBootTest
@Testcontainers
public class BorrowConcurrencyTest {

    @Container
    public static MySQLContainer<?> mysql = new MySQLContainer<>("mysql:8.0")
            .withDatabaseName("testdb")
            .withUsername("test")
            .withPassword("test");

    @Autowired
    private BookMapper bookMapper;
    @Autowired
    private BookCopyMapper copyMapper;
    @Autowired
    private BorrowService borrowService;

    @BeforeAll
    public static void setUpAll() {
        // testcontainers will expose JDBC URL via env; Spring Boot can be configured to use it via properties override in test profile
        System.setProperty("spring.datasource.url", mysql.getJdbcUrl());
        System.setProperty("spring.datasource.username", mysql.getUsername());
        System.setProperty("spring.datasource.password", mysql.getPassword());
    }

    @Test
    public void testConcurrentBorrow() throws InterruptedException {
        // 1. create book
        Book b = new Book();
        b.setTitle("并发测试书");
        b.setTotalCopies(1);
        b.setAvailableCopies(1);
        bookMapper.insert(b);
        Long bookId = b.getId();
        // 2. create one copy
        BookCopy c = new BookCopy();
        c.setBookId(bookId);
        c.setBarcode("C1-" + System.currentTimeMillis());
        c.setCopyStatus(1);
        copyMapper.insert(c); // you may need insert method in mapper for tests; or insert via SQL
        // 3. run two concurrent borrow attempts
        ExecutorService es = Executors.newFixedThreadPool(2);
        CountDownLatch start = new CountDownLatch(1);
        CountDownLatch done = new CountDownLatch(2);
        AtomicInteger success = new AtomicInteger(0);
        Runnable task = () -> {
            try {
                start.await();
                Long rec = borrowService.borrowBook(ThreadLocalRandom.current().nextLong(1000, 2000), bookId);
                if (rec != null) success.incrementAndGet();
            } catch (Exception e) {
                e.printStackTrace();
            } finally {
                done.countDown();
            }
        };
        es.submit(task);
        es.submit(task);
        start.countDown();
        done.await();
        Assertions.assertEquals(1, success.get(), "only one borrow should succeed");
    }
}
```

## 8) 改进与扩展（面试可讲）

- 优化：当前是**悲观锁（FOR UPDATE）**，适合写频率不高但要求强一致的场景。若读多写少或高并发可采用 Redis 预减库存 + 后端异步校验（最终一致性）。
- 预约（reservation）：当没有可借副本时，将用户加入 reservation table。还书时触发通知并将副本保留给队首一定时间。
- 超期和罚金：定时任务扫描 `borrow_records` 设置过期状态并通知用户。
- 性能：确保 `SELECT ... FOR UPDATE` 的 WHERE 子句走索引，避免全表扫描和锁膨胀。
- 测试覆盖：写更多集成测试（多用户、多副本场景），以及 failover 场景（事务回滚、异常中断）。
