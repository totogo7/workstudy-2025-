# 二、关键原则（实现时记住）

- 前后端约定：`Authorization: Bearer <token>`；后端 token 来自 `/api/auth/login`。实际生产推荐 httpOnly cookie，但 demo 用 localStorage/Pinia 方便演示。
- 分页、错误统一处理在 axios 拦截器里做（401 自动跳转登录）。
- 路由守卫：未登录访问受保护路由重定向到 `/login`。
- 组件化、复用：BookCard、BookEdit 等复用。
- UX：加载态、提示、表单校验（Element 的 `rules`）。
- 易测试：组件的业务逻辑分离到 `api/*` 和 `store`，方便单测/集成测试。

### `src/api/index.js` — axios 封装 & 拦截器

```js
import axiosLib from 'axios';
import { useAuthStore } from '@/store';

const axios = axiosLib.create({
  baseURL: 'http://localhost:8080',
  timeout: 10000,
});

// request interceptor: attach token
axios.interceptors.request.use((config) => {
  const store = useAuthStore();
  if (store.token) {
    config.headers.Authorization = `Bearer ${store.token}`;
  }
  return config;
}, (error) => Promise.reject(error));

// response interceptor: global error handling
axios.interceptors.response.use(res => res, error => {
  const status = error?.response?.status;
  if (status === 401) {
    // logout and redirect to login
    const store = useAuthStore();
    store.logout();
    window.location.href = '/login';
  }
  return Promise.reject(error);
});

export default axios;
```

### `src/store/index.js` — Pinia auth store

```js
import { defineStore } from 'pinia';
import axios from '@/api';

export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: localStorage.getItem('token') || null,
    username: localStorage.getItem('username') || null,
  }),
  actions: {
    setToken(token, username) {
      this.token = token;
      this.username = username;
      localStorage.setItem('token', token);
      localStorage.setItem('username', username);
    },
    logout() {
      this.token = null;
      this.username = null;
      localStorage.removeItem('token');
      localStorage.removeItem('username');
    },
    async login(username, password) {
      const res = await axios.post('/api/auth/login', { username, password });
      if (res.data.code === 1) {
        this.setToken(res.data.data.token, username);
        return true;
      } else {
        throw new Error(res.data.msg);
      }
    }
  }
});
```

### `src/router/index.js` — 路由 + 守卫

```js
import { createRouter, createWebHistory } from 'vue-router';
import Login from '@/views/Login.vue';
import Home from '@/views/Home.vue';
import BookList from '@/views/BookList.vue';
import BookDetail from '@/views/BookDetail.vue';
import BorrowRecords from '@/views/BorrowRecords.vue';
import AdminBooks from '@/views/AdminBooks.vue';
import { useAuthStore } from '@/store';

const routes = [
  { path: '/', name: 'Home', component: Home },
  { path: '/login', name: 'Login', component: Login },
  { path: '/books', name: 'Books', component: BookList },
  { path: '/books/:id', name: 'BookDetail', component: BookDetail },
  { path: '/my/borrow', name: 'BorrowRecords', component: BorrowRecords, meta: { requiresAuth: true } },
  { path: '/admin/books', name: 'AdminBooks', component: AdminBooks, meta: { requiresAuth: true, requiresAdmin: true } }
];

const router = createRouter({
  history: createWebHistory(),
  routes
});

router.beforeEach((to, from, next) => {
  const auth = useAuthStore();
  if (to.meta.requiresAuth && !auth.token) {
    return next({ name: 'Login', query: { redirect: to.fullPath }});
  }
  // optional: check admin role from token payload if you stored role in store
  next();
});

export default router;
```

<template>
  <div class="login-page">
    <el-card class="login-card">
      <h2>登录</h2>
      <el-form :model="form" :rules="rules" ref="formRef">
        <el-form-item label="用户名" prop="username"><el-input v-model="form.username" /></el-form-item>
        <el-form-item label="密码" prop="password"><el-input type="password" v-model="form.password" /></el-form-item>
        <el-form-item>
          <el-button type="primary" @click="onLogin" :loading="loading">登录</el-button>
        </el-form-item>
      </el-form>
    </el-card>
  </div>
</template>

<script>
import { ref } from 'vue';
import { useAuthStore } from '@/store';
export default {
  setup() {
    const formRef = ref(null);
    const form = ref({ username: '', password: '' });
    const loading = ref(false);
    const auth = useAuthStore();
    const rules = {
      username: [{ required: true, message: '请输入用户名', trigger: 'blur' }],
      password: [{ required: true, message: '请输入密码', trigger: 'blur' }]
    };
    const onLogin = async () => {
      formRef.value.validate(async (valid) => {
        if (!valid) return;
        loading.value = true;
        try {
          await auth.login(form.value.username, form.value.password);
          // redirect if present
          const redirect = new URL(window.location).searchParams.get('redirect') || '/';
          window.location.href = redirect;
        } catch (err) {
          ElMessage.error(err.message || '登录失败');
        } finally { loading.value = false; }
      });
    };
    return { formRef, form, rules, onLogin, loading };
  }
}
</script>

  listen 80;
  server_name yourdomain.com;
  root /usr/share/nginx/html;
